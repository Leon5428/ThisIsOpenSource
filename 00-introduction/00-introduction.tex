\begin{frame}{为什么我们需要操作系统}
\begin{block}{如果你正在使用电脑...}
\hspace*{2em}这时，你的导员在QQ上给你发了文件，你的部长在微信上给你发了通知，你的同学在抖音上给你转发了视频...那处理器该先给谁用呢？内存该怎么分配呢？

\par\hspace*{2em}我们既想让它们运行又不想让它们挤破头，该怎么办呢？
\end{block}

\pause
\begin{block}{我们需要一个管理者！}
\hspace*{2em}就像厨房需要厨师长安排厨师的工作、分配厨具的使用一样，计算机也需要一个管理者，负责协调各个程序对硬件资源的使用。

\par\hspace*{2em}它要安排程序何时使用处理器、合理分配内存，还要尽可能避免不同程序之间的相互干扰。

\par\hspace*{2em}\alert{这个管理者，就是操作系统（Operating System，OS）。}
\end{block}
\end{frame}

\begin{frame}{什么是OS}
	\begin{block}{操作系统（Operating System，OS）}
		\hspace*{2em}操作系统是管理计算机硬件与软件资源，并为应用程序提供运行环境和基础服务的系统软件。
	\end{block}
		
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.48\textwidth}
			\begin{block}{资源管理者}
				\begin{itemize}
					\item 安排处理器工作
					\item 分配内存空间
					\item 管理文件与设备
				\end{itemize}
			\end{block}
		\end{column}
			
		\begin{column}{0.48\textwidth}
			\begin{block}{程序服务者}
				\begin{itemize}
					\item 帮助程序读写文件
					\item 提供统一的设备接口
					\item 为程序提供运行环境
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}
		
	\begin{center}
		\alert{操作系统既是资源的管理者，也是连接应用程序与计算机硬件的桥梁。}
	\end{center}
\end{frame}


\begin{frame}{为什么会是程序服务者？}
	
	\begin{block}{当有人在微信上给你发送了一张图片}
		\hspace*{2em}当你点击“保存”时，程序需要将图片保存到硬盘上。但程序应该怎样操作硬盘呢？
	\end{block}
	
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.48\textwidth}
			\begin{block}{没有统一的系统服务}
				程序需要自行处理存储设备的
				操作细节，甚至适配不同的硬件。
			\end{block}
		\end{column}
		
		\begin{column}{0.48\textwidth}
			\begin{block}{有操作系统提供服务}
				程序只需要请求“写入文件”，
				操作系统就可以帮助完成底层操作。
			\end{block}
		\end{column}
	\end{columns}
	
	\begin{block}{保存图片的过程（简化）}
		\centering
		微信
		$\longrightarrow$
		操作系统
		$\longrightarrow$
		存储设备
	\end{block}
	
	\vspace{0.3cm}
	
	\begin{center}
		\alert{操作系统为应用程序提供统一的服务，
			让程序不必直接处理复杂的硬件细节。}
	\end{center}
	
\end{frame}

\begin{frame}{操作系统发展史}
	
	\begin{block}{真的需要那么多人工参与吗？}
		\hspace*{2em}假设我们有 100 个计算任务，
		如果每完成一个任务，都需要工作人员
		手动准备下一个任务，不仅麻烦，
		还会让昂贵的计算机白白等待。
	\end{block}
	
	\vspace{0.3cm}
	
	\begin{block}{能不能让计算机自己工作呢？}
		\hspace*{2em}能不能提前把任务准备好，
		让计算机完成一个任务后，
		自动开始执行下一个任务呢？
		
		\centering
		任务一 $\longrightarrow$ 任务二
		$\longrightarrow$ 任务三
		$\longrightarrow$ $\cdots$
	\end{block}
	
	\begin{block}{单道批处理系统（Simple Batch System）}
		\hspace*{2em}20 世纪 50 年代，
		批处理系统逐渐出现。
		它能够自动组织并依次执行一批作业，
		减少人工干预，提高计算机的使用效率，但此时的批处理系统，还是一次只运行一次作业。
	\end{block}
	
\end{frame}
